//! Smoke-test the exact .smod against the Sunrise Wasmtime loader.
//! This intentionally validates package loading and initialization, not gameplay.
#[path = "support/no_hooks.rs"]
mod no_hooks;
#[path = "../src/test_stack.rs"]
mod test_stack;

use sunrise_mod_package::manifest::Package;
use sunrise_mod_wasm::{WasmPackage, load};
use std::path::Path;

#[test]
fn sunrise_qol_loads_and_initializes() {
    test_stack::on_guest_stack(|| {
        let package_path = std::env::var("SUNRISE_QOL_SMOD").expect("built .smod");
        let standard_path = std::env::var("SUNRISE_WASM_STD").expect("matching WASM standard");
        let path = Path::new(&package_path);
        let parsed = Package::read(path).expect("valid package manifest and hashes");
        let manifest = parsed.manifest;
        assert_eq!(manifest.id, "horizon_qol_suite");
        assert!(manifest.inject.is_empty(), "the mod must use the published API");
        let info = mod_api::PackageInfo::new(
            manifest.id.clone(),
            manifest.version.to_string(),
            mod_api::PackageKind::Code,
            mod_api::Side::Both,
            vec![manifest.id.clone()],
            vec![],
        );
        let standard = std::fs::read(standard_path).expect("WASM standard");
        let guest = Package::read_wasm(path).expect("mod.wasm");
        mod_api::__private::initialize().expect("initialize Sunrise host API");
        let loaded = load(&standard, vec![WasmPackage {
            manifest,
            module: guest,
        }]);
        assert!(loaded.refused.is_empty(), "Sunrise refused mod: {:?}", loaded.refused);
        assert_eq!(loaded.mods.len(), 1, "the package must load");
        loaded.initialize(&[info]).expect("Sunrise guest initialization");
    });
}
