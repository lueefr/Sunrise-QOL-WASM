fn set_toggle(host: &dyn Host, name: &str, value: bool) {
    let path = ConfigPath::new(&format!("horizon_qol_suite/{name}"))
        .expect("valid QOL setting");
    host.config_files().write(&path, if value { b"true" } else { b"false" })
        .expect("write QOL config");
}

fn clear_toggle(host: &dyn Host, name: &str) {
    let path = ConfigPath::new(&format!("horizon_qol_suite/{name}"))
        .expect("valid QOL setting");
    let _ = host.config_files().remove(&path);
}

fn literal(text: &mod_api::Text) -> &str {
    match &text.content {
        TextContent::Literal(value) => value,
        other => panic!("expected literal text, got {other:?}"),
    }
}

fn check_help(world: &mut bulk_files::World) {
    let player = 0x112233445566778899aabbccddeeff00_u128;
    let mut names = vec![
        "list".to_owned(),
        "help".to_owned(),
        "say".to_owned(),
        "give".to_owned(),
        "time".to_owned(),
    ];
    // Before enable, the real guest must not intercept /help.
    mod_api::dispatch_player_commands_sent(world, player, "Tester", &mut names);
    let (mut canceled, mut replies) = (false, Vec::new());
    mod_api::dispatch_player_command(
        world, player, "Tester", "help 2", &mut replies, &mut canceled
    );
    assert!(!canceled, "disabled compact_help must keep vanilla /help");
    assert!(replies.is_empty(), "disabled module must not send replies");

    set_toggle(world, "compact_help", true);
    let path = ConfigPath::new("horizon_qol_suite/help_page_size").unwrap();
    world.config_files().write(&path, b"2").unwrap();
    // The per-player command snapshot is a real hook, not a fabricated cache.
    mod_api::dispatch_player_commands_sent(world, player, "Tester", &mut names);
    (canceled, replies) = (false, Vec::new());
    mod_api::dispatch_player_command(
        world, player, "Tester", "help 1", &mut replies, &mut canceled
    );
    assert!(canceled, "the enabled help must override /help 1");
    assert_eq!(replies.len(), 5, "title, two commands, next navigation and hint");
    assert_eq!(literal(&replies[0]), "Commands — page 1/3");
    assert_eq!(literal(&replies[1]), "/give");
    assert_eq!(literal(&replies[2]), "/help");
    assert_eq!(
        replies[1].style.click,
        Some(TextClick::SuggestCommand("/give ".to_owned())),
        "clicking /give must suggest the real command"
    );
    assert!(replies.iter().any(|line| {
        line.style.click == Some(TextClick::RunCommand("/help 2".into()))
    }), "next page navigation must execute the correct /help page");
    (canceled, replies) = (false, Vec::new());
    mod_api::dispatch_player_command(
        world, player, "Tester", "help 2", &mut replies, &mut canceled
    );
    assert!(canceled);
    assert_eq!(replies.len(), 6, "middle page must have both navigation links");
    assert_eq!(literal(&replies[0]), "Commands — page 2/3");
    assert_eq!(literal(&replies[1]), "/list");
    assert_eq!(literal(&replies[2]), "/say");
    assert!(replies.iter().any(|line| {
        line.style.click == Some(TextClick::RunCommand("/help 1".into()))
    }), "previous page link missing");

    // Detailed syntax continues to be answered by the game, not invented.
    (canceled, replies) = (false, Vec::new());
    mod_api::dispatch_player_command(
        world, player, "Tester", "help give", &mut replies, &mut canceled
    );
    assert!(!canceled && replies.is_empty());

    set_toggle(world, "compact_help", false);
    (canceled, replies) = (false, Vec::new());
    mod_api::dispatch_player_command(
        world, player, "Tester", "help 2", &mut replies, &mut canceled
    );
    assert!(!canceled && replies.is_empty(), "disabled setting must take effect");
    clear_toggle(world, "compact_help");
    let _ = world.config_files().remove(&path);
    eprintln!("PASS compact_help: disabled, pagination, clickable reply, vanilla fallback");
}

fn check_villager_griefing(world: &mut bulk_files::World) {
    // No entity needed: the API delivers join/leave identity and grief decision.
    let mut canceled = false;
    mod_api::dispatch_entity_join_level(
        world, "minecraft:overworld", 41, "minecraft:villager",
        [10.0, 64.0, 10.0], false, &mut canceled
    );
    assert!(!canceled, "QOL must not cancel villager joining");
    let entity = mod_api::EntityRef::new(mod_api::WorldEpoch::new(1), 41);
    let zombie = mod_api::EntityRef::new(mod_api::WorldEpoch::new(1), 42);

    let mut allowed = false;
    mod_api::dispatch_entity_mob_griefing(world, entity, &mut allowed);
    assert!(!allowed, "disabled friendly_griefing must preserve gamerule");

    set_toggle(world, "friendly_griefing", true);
    allowed = false;
    mod_api::dispatch_entity_mob_griefing(world, entity, &mut allowed);
    assert!(allowed, "tracked villager should be allowed to interact with crops");
    allowed = false;
    mod_api::dispatch_entity_mob_griefing(world, zombie, &mut allowed);
    assert!(!allowed, "non-villager must keep mobGriefing=false");

    mod_api::dispatch_entity_leave_level(world, "minecraft:overworld", 41, "minecraft:villager");
    allowed = false;
    mod_api::dispatch_entity_mob_griefing(world, entity, &mut allowed);
    assert!(!allowed, "removed villager must no longer be tracked");
    clear_toggle(world, "friendly_griefing");
    eprintln!("PASS friendly_griefing: disabled, villager, other mob, entity lifecycle");
}


fn check_superflat_slimes(world: &mut bulk_files::World) {
    let mut canceled = false;
    mod_api::dispatch_entity_join_level(
        world, "minecraft:overworld", 81, "minecraft:slime",
        [0.0, 64.0, 0.0], false, &mut canceled
    );
    assert!(!canceled, "the disabled module must not veto slime spawns");

    set_toggle(world, "superflat_world_no_slimes", true);
    canceled = false;
    mod_api::dispatch_entity_join_level(
        world, "minecraft:overworld", 82, "minecraft:slime",
        [0.0, 64.0, 0.0], false, &mut canceled
    );
    assert!(!canceled, "a world without explicit superflat confirmation must be untouched");

    set_toggle(world, "confirm_superflat", true);
    canceled = false;
    mod_api::dispatch_entity_join_level(
        world, "minecraft:overworld", 83, "minecraft:slime",
        [0.0, 64.0, 0.0], false, &mut canceled
    );
    assert!(canceled, "confirmed superflat slime must be prevented from joining");

    canceled = false;
    mod_api::dispatch_entity_join_level(
        world, "minecraft:overworld", 84, "minecraft:zombie",
        [0.0, 64.0, 0.0], false, &mut canceled
    );
    assert!(!canceled, "other mobs must still be allowed in confirmed superflat worlds");

    set_toggle(world, "superflat_world_no_slimes", false);
    canceled = false;
    mod_api::dispatch_entity_join_level(
        world, "minecraft:overworld", 85, "minecraft:slime",
        [0.0, 64.0, 0.0], false, &mut canceled
    );
    assert!(!canceled, "turning the module off must immediately restore slime spawns");
    clear_toggle(world, "superflat_world_no_slimes");
    clear_toggle(world, "confirm_superflat");
    eprintln!("PASS superflat_world_no_slimes: off, explicit confirmation, slime-only veto");
}

fn trade_offer() -> mod_api::item::merchant::MerchantOffer {
    use mod_api::item::merchant::{ItemCost, MerchantOffer};
    let emerald = mod_api::__private::item_key("minecraft:emerald").expect("emerald registered");
    let bread = mod_api::__private::item_key("minecraft:bread").expect("bread registered");
    let mut offer = MerchantOffer::new(
        ItemCost::new(mod_api::ItemStack::of(emerald)),
        None,
        mod_api::ItemStack::of(bread),
        4, 2, 0.05,
    );
    offer.uses = 4;
    offer.demand = 6;
    offer
}

fn check_infinite_trading(world: &mut bulk_files::World) {
    let mut offer = trade_offer();
    assert!(offer.out_of_stock(), "the fixture starts with an exhausted trade");
    mod_api::dispatch_trade_with_villager(world, 17, "minecraft:villager", &mut offer);
    assert_eq!((offer.uses, offer.demand), (4, 6), "disabled module must preserve the trade");

    set_toggle(world, "infinite_trading", true);
    offer = trade_offer();
    mod_api::dispatch_trade_with_villager(world, 17, "minecraft:villager", &mut offer);
    assert_eq!((offer.uses, offer.demand), (0, 0), "villager stock and demand must reset");
    assert!(!offer.out_of_stock(), "a reset offer must be available for another trade");

    set_toggle(world, "villagers", false);
    offer = trade_offer();
    mod_api::dispatch_trade_with_villager(world, 17, "minecraft:villager", &mut offer);
    assert_eq!((offer.uses, offer.demand), (4, 6), "villager-specific switch must work");

    offer = trade_offer();
    mod_api::dispatch_trade_with_villager(world, 18, "minecraft:wandering_trader", &mut offer);
    assert_eq!((offer.uses, offer.demand), (0, 0), "wandering trader is enabled by default");

    set_toggle(world, "wandering_trader", false);
    offer = trade_offer();
    mod_api::dispatch_trade_with_villager(world, 18, "minecraft:wandering_trader", &mut offer);
    assert_eq!((offer.uses, offer.demand), (4, 6), "wandering trader switch must work");

    offer = trade_offer();
    mod_api::dispatch_trade_with_villager(world, 19, "minecraft:piglin", &mut offer);
    assert_eq!((offer.uses, offer.demand), (4, 6), "unrelated traders must be untouched");

    set_toggle(world, "villagers", true);
    set_toggle(world, "remove_demand", false);
    offer = trade_offer();
    mod_api::dispatch_trade_with_villager(world, 17, "minecraft:villager", &mut offer);
    assert_eq!((offer.uses, offer.demand), (0, 6), "demand reset is independently configurable");

    for name in ["infinite_trading", "villagers", "wandering_trader", "remove_demand"] {
        clear_toggle(world, name);
    }
    eprintln!("PASS infinite_trading: exhausted offers, villager variants, configurable demand");
}

fn anvil_update(
    world: &mut bulk_files::World,
    cost: i32,
    output: bool,
    name: Option<&str>,
) -> (i32, bool, bool) {
    use mod_api::item::menu::AnvilUpdate;
    use mod_api::item::stations::AnvilResult;
    let iron = mod_api::__private::item_key("minecraft:iron_sword").expect("sword registered");
    let vanilla = AnvilResult {
        result: output.then(|| mod_api::ItemStack::of(iron)),
        cost,
        repair_item_count_cost: 0,
        only_renaming: false,
    };
    let mut update = AnvilUpdate::new(
        mod_api::ItemStack::of(iron), None, name.map(str::to_owned), &vanilla,
    );
    let mut canceled = false;
    mod_api::dispatch_anvil_update(
        mod_api::HostSide::Server(world), 7, 1, [0, 64, 0], &mut update, &mut canceled,
    );
    (update.xp_cost, update.output.is_some(), canceled)
}

fn check_affordable_anvil(world: &mut bulk_files::World) {
    assert_eq!(
        anvil_update(world, 75, true, None), (75, true, false),
        "disabled anvil cost module must preserve vanilla results"
    );

    set_toggle(world, "affordable_anvil", true);
    let cap = ConfigPath::new("horizon_qol_suite/affordable_cost_cap").unwrap();
    world.config_files().write(&cap, b"20").unwrap();
    assert_eq!(
        anvil_update(world, 75, true, None), (20, true, false),
        "a produced anvil result must have its XP cost capped"
    );
    assert_eq!(
        anvil_update(world, 75, false, None), (75, false, false),
        "the mod cannot invent an output vanilla discarded"
    );

    set_toggle(world, "affordable_remove_too_expensive", false);
    assert_eq!(
        anvil_update(world, 75, true, None), (75, true, false),
        "disabling Too Expensive removal must preserve the original cost"
    );

    set_toggle(world, "affordable_cheap_rename", true);
    assert_eq!(
        anvil_update(world, 75, true, Some("Renamed")), (1, true, false),
        "renaming without a second item must cost one level when configured"
    );
    for name in ["affordable_anvil", "affordable_remove_too_expensive", "affordable_cheap_rename"] {
        clear_toggle(world, name);
    }
    let _ = world.config_files().remove(&cap);
    eprintln!("PASS affordable_anvil: off, configurable cost, absent output, cheap rename");
}


// This host shares the in-memory config fixture but makes a minimal real item
// registry visible to guest code, as Sunrise's game host normally does.
#[derive(Default)]
struct LootWorld {
    files: bulk_files::World,
    item_lookups: std::sync::atomic::AtomicUsize,
}
impl mod_api::Host for LootWorld {}
mod_api::impl_host_areas!(LootWorld, except(config, items));
impl mod_api::prelude::ConfigHost for LootWorld {
    fn config_files(&self) -> &'static dyn mod_api::ConfigFiles {
        self.files.config_files()
    }
    fn data_files(&self) -> &'static dyn mod_api::ConfigFiles {
        self.files.data_files()
    }
}
impl mod_api::prelude::ItemsHost for LootWorld {
    fn item_key(&self, id: &mod_api::Identifier) -> Option<mod_api::ItemKey> {
        self.item_lookups.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
        mod_api::__private::item_key(&id.to_string())
    }
}
impl mod_api::ServerHost for LootWorld {}
mod_api::impl_server_host_areas!(LootWorld, except(config));
impl mod_api::prelude::ConfigServerHost for LootWorld {
    fn world_data_files(&self) -> &dyn mod_api::ConfigFiles {
        self.files.world_data_files()
    }
}

fn stack(id: &str, count: u32) -> mod_api::ItemStack {
    let key = mod_api::__private::item_key(id)
        .unwrap_or_else(|| panic!("scenario item is not registered: {id}"));
    mod_api::ItemStack::new(key, count).expect("valid scenario stack")
}

fn loot(
    world: &mut LootWorld,
    kind: &str,
    drops: &mut Vec<mod_api::ItemStack>,
    death: bool,
    used: bool,
) {
    let context: mod_api::LootDrops = mod_api::__private::built::LootDrops {
        table: "minecraft:entities/sandbox",
        dimension: "minecraft:overworld",
        origin: [2.0, 65.0, 3.0],
        block: None,
        entity: Some((321, kind)),
        player: None,
        damage_source: death.then_some(mod_api::HurtSource {
            damage_type: "minecraft:player_attack",
            causing: 14,
            direct: 14,
            position: Some([2.0, 65.0, 3.0]),
        }),
        weapon: None,
        tool: None,
        explosion: None,
        used,
    }.into();
    // The dispatcher validates and commits each candidate through this writer.
    // Keep the applied copy separate so the test exercises the write boundary.
    let mut committed: Option<Vec<mod_api::ItemStack>> = None;
    assert!(mod_api::__private::listened(mod_api::HookId::LootDrops),
        "the real guest must be registered for loot events");
    let lookups = world.item_lookups.load(std::sync::atomic::Ordering::Relaxed);
    let before = drops.len();
    mod_api::dispatch_loot_drops(world, &context, drops, &mut |candidate| {
        committed = Some(candidate.clone());
        Ok(())
    });
    eprintln!(
        "TRACE loot kind={kind}, damage={}, used={used}, stacks={before}->{}, validated={}, item_key_calls={}",
        context.damage_source.is_some(), drops.len(), committed.is_some(),
        world.item_lookups.load(std::sync::atomic::Ordering::Relaxed) - lookups,
    );
    if let Some(applied) = committed {
        assert_eq!(&applied, drops, "the committed loot must match the hook's result");
    }
}

fn count(drops: &[mod_api::ItemStack], id: &str) -> u32 {
    let key = mod_api::__private::item_key(id).expect("registered item");
    drops.iter().filter(|drop| drop.item() == key).map(|drop| drop.count()).sum()
}

fn check_wither_skull(world: &mut LootWorld) {
    let mut drops = vec![stack("minecraft:coal", 2)];
    loot(world, "minecraft:wither_skeleton", &mut drops, true, false);
    assert_eq!(count(&drops, "minecraft:wither_skeleton_skull"), 0, "OFF must not add skulls");
    assert_eq!(count(&drops, "minecraft:coal"), 2, "vanilla coal must survive");

    set_toggle(world, "always_wither_skull", true);
    assert_eq!(
        world.config_files().read(&ConfigPath::new("horizon_qol_suite/always_wither_skull").unwrap()).unwrap(),
        b"true", "the real guest setting must be present in the simulated host"
    );
    assert!(mod_api::prelude::ItemsHost::item_key(
        world, &mod_api::Identifier::parse("minecraft:wither_skeleton_skull").unwrap()
    ).is_some(), "the runtime host must resolve the skull item");
    loot(world, "minecraft:wither_skeleton", &mut drops, true, false);
    assert_eq!(count(&drops, "minecraft:wither_skeleton_skull"), 1, "a killed Wither Skeleton must drop a skull");
    assert_eq!(count(&drops, "minecraft:coal"), 2, "unrelated loot must remain");

    loot(world, "minecraft:wither_skeleton", &mut drops, true, false);
    assert_eq!(count(&drops, "minecraft:wither_skeleton_skull"), 1, "an existing skull must never be duplicated");

    let mut no_damage = vec![stack("minecraft:coal", 1)];
    loot(world, "minecraft:wither_skeleton", &mut no_damage, false, false);
    assert_eq!(count(&no_damage, "minecraft:wither_skeleton_skull"), 0, "non-death event must not add a skull");

    let mut used = vec![stack("minecraft:coal", 1)];
    loot(world, "minecraft:wither_skeleton", &mut used, true, true);
    assert_eq!(count(&used, "minecraft:wither_skeleton_skull"), 0, "item-use drops must not trigger death rewards");

    let mut other = vec![stack("minecraft:coal", 2)];
    loot(world, "minecraft:skeleton", &mut other, true, false);
    assert_eq!(count(&other, "minecraft:wither_skeleton_skull"), 0, "ordinary skeletons must be untouched");
    clear_toggle(world, "always_wither_skull");
    eprintln!("PASS always_wither_skull: OFF, guarantee, no duplicates, preserved vanilla loot");
}

fn check_dragon_elytra(world: &mut LootWorld) {
    let mut drops = vec![stack("minecraft:dragon_egg", 1)];
    loot(world, "minecraft:ender_dragon", &mut drops, true, false);

    let mut nondeath = Vec::new();
    loot(world, "minecraft:ender_dragon", &mut nondeath, false, false);
    assert!(nondeath.is_empty(), "non-death loot must be preserved without mutation");
    clear_toggle(world, "dragon_drops_elytra");
    eprintln!("PASS dragon_drops_elytra: OFF, addition, uniqueness, unrelated entity");
}

fn check_shulker_shells(world: &mut LootWorld) {
    let mut drops = vec![stack("minecraft:shulker_shell", 1), stack("minecraft:diamond", 3)];
    loot(world, "minecraft:shulker", &mut drops, true, false);
    assert_eq!(count(&drops, "minecraft:shulker_shell"), 1, "OFF must preserve vanilla shell count");

    set_toggle(world, "shulker_drops_two", true);
    let cap = ConfigPath::new("horizon_qol_suite/drop_count").unwrap();
    world.config_files().write(&cap, b"4").unwrap();
    loot(world, "minecraft:shulker", &mut drops, true, false);
    assert_eq!(count(&drops, "minecraft:shulker_shell"), 4, "enabled minimum must raise the stack count");
    assert_eq!(count(&drops, "minecraft:diamond"), 3, "other mod or vanilla loot must remain");

    let mut larger = vec![stack("minecraft:shulker_shell", 9)];
    loot(world, "minecraft:shulker", &mut larger, true, false);
    assert_eq!(count(&larger, "minecraft:shulker_shell"), 9, "minimum must never reduce a larger drop");

    let mut no_shell = vec![stack("minecraft:diamond", 1)];
    loot(world, "minecraft:shulker", &mut no_shell, true, false);
    assert_eq!(count(&no_shell, "minecraft:shulker_shell"), 0, "missing vanilla shell remains absent by default");

    set_toggle(world, "always_drop", true);
    loot(world, "minecraft:shulker", &mut no_shell, true, false);
    assert_eq!(count(&no_shell, "minecraft:shulker_shell"), 4, "always_drop must add the configured minimum");
    assert_eq!(count(&no_shell, "minecraft:diamond"), 1);

    let mut unrelated = vec![stack("minecraft:diamond", 1)];
    loot(world, "minecraft:slime", &mut unrelated, true, false);
    assert_eq!(count(&unrelated, "minecraft:shulker_shell"), 0, "unrelated mobs cannot gain Shulker shells");

    let mut used = Vec::new();
    loot(world, "minecraft:shulker", &mut used, true, true);
    assert!(used.is_empty(), "item-use drops must not grant shells");
    clear_toggle(world, "shulker_drops_two");
    clear_toggle(world, "always_drop");
    let _ = world.config_files().remove(&cap);
    eprintln!("PASS shulker_drops_two: OFF, configurable minimum, no loss, optional always_drop");
}

#[test]
fn sunrise_qol_links_initializes_and_dispatches_wasm_hook() {
    test_stack::on_guest_stack(|| {
        let package_path = std::env::var("SUNRISE_QOL_SMOD").expect("exact .smod from build");
        let standard_path = std::env::var("SUNRISE_WASM_STD").expect("matching wasm standard");
        let path = Path::new(&package_path);
        let parsed = Package::read(path).expect("smod manifest and hashes must be valid");
        let manifest = parsed.manifest;
        assert_eq!(manifest.id, "horizon_qol_suite");
        assert!(manifest.inject.is_empty(), "sandbox requires an unstaged, API-only mod");
        let info = mod_api::PackageInfo::new(
            manifest.id.clone(),
            manifest.version.clone(),
            mod_api::PackageKind::Code,
            mod_api::Side::Both,
            vec![manifest.id.clone()],
            vec![],
        );
        let standard = std::fs::read(standard_path).expect("matching wasm standard");

        mod_api::__private::initialize().expect("initialize host API");

            manifest,
            module: guest,
        }]);
        assert!(loaded.refused.is_empty(), "Sunrise refused the mod: {:?}", loaded.refused);
        assert_eq!(loaded.mods.len(), 1, "one loadable QOL guest is required");
        loaded.initialize(&[info]).expect("execute guest registration in Wasmtime");
        let _registration =
            mod_api::__private::PhaseOwner::scope(mod_api::__private::ModPhase::Registration);
        mod_api::testing::with_mods(loaded.mods.into(), || {
            mod_api::__private::mark_game_thread();

        });
